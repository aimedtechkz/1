				showNotification('❌ Ошибка отправки. Пожалуйста, попробуйте еще раз или свяжитесь по телефону.', 'error');
			} finally {
				// Восстанавливаем кнопку
				submitButton.textContent = originalText;
				submitButton.disabled = false;
			}
		});
       
        // Lazy loading покрывается нативным loading="lazy" в HTML. JS-polyfill не нужен для современных браузеров.

        /* Исправленный Fallback для сломанных img в продуктах */
        document.addEventListener('DOMContentLoaded', () => {
            const handleImageError = (img) => {
                if (img.naturalWidth === 0) {
                    const icon = img.nextElementSibling;
                    if (icon && icon.tagName === 'I') {
                        img.style.display = 'none';
                        icon.style.opacity = '1';
                        icon.style.fontSize = '3rem';
                        img.alt = img.alt + ' (недоступно)';
                    }
                }
            };
          
            document.querySelectorAll('.product-image img').forEach(img => {
                img.addEventListener('error', () => handleImageError(img));
                img.addEventListener('load', () => handleImageError(img));
                if (img.complete) {
                    handleImageError(img);
                }
            });

            // Автоматически показываем все товары при загрузке страницы
            const allButton = document.querySelector('.filter-btn[data-category="all"]');
            if (allButton) {
                allButton.click();  // Это сработает, потому что у вас уже есть обработчик на клик
            }
        });
    </script>
